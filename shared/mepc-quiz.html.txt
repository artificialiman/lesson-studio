function renderQ(idx){
  cur = idx;
  var q = questions[idx];
  var html = '';

  // Passage block (comprehension)
  if (q.passage){
    html += '<div class="passage-box"><div class="passage-label"><i class="fas fa-book-open"></i> Read the passage below, then answer the questions</div>' + q.passage + '</div>';
  }

  // Diagram block
  if (q.diagram){
    html += '<div class="diagram-box">' + q.diagram + (q.diagramCaption ? '<div class="diagram-caption">'+q.diagramCaption+'</div>' : '') + '</div>';
  }

  // Question text
  html += '<div class="qnum">Question '+(idx+1)+' of '+questions.length+(q.subject?' &bull; '+q.subject:'')+(q.topic?' &bull; '+q.topic:'')+'</div>';
  html += '<div class="qtext">'+q.text+'</div>';

  // Options
  html += '<div class="options">';

    var txt = q['option'+l];
    if (!txt) return;
    var cls = 'opt';
    if (userAnswers[idx]===l) cls += ' selected';

      if (l===q.answer) cls += ' correct-review';

      '<div class="olabel">'+l+'</div>' +
      '<div class="otext">'+txt+'</div></div>';
  });
  html += '</div>';

  // Explanation (review mode only)
  if (reviewMode && (q.explanation || q.exception)){
    html += '<div class="expl show">';
    if (q.explanation) html += '<div class="expl-head"><i class="fas fa-lightbulb"></i> Explanation</div><div>'+q.explanation+'</div>';
    if (q.exception)   html += '<div class="note"><i class="fas fa-exclamation-circle"></i> Note: '+q.exception+'</div>';
    html += '</div>';
  }

  html += '<div class="qnav">';
  html += '<button class="btn btn-secondary" onclick="goTo('+(idx-1)+')" '+(idx===0?'disabled':'')+'><i class="fas fa-arrow-left"></i> Prev</button>';

// ── Scoring ──────────────────────────────────────────────
function computeScore(){
  var correct=0, wrong=0, skipped=0, jamb=0;
  questions.forEach(function(q,i){
    var a = userAnswers[i];
    if (!a){ skipped++; }
    else if (a===q.answer){ correct++; jamb+=SC_CORRECT; }
    else { wrong++; jamb+=SC_WRONG; }
  });
  jamb = Math.max(0, jamb);
  var rawPct = questions.length ? correct/questions.length*100 : 0;
  return { correct:correct, wrong:wrong, skipped:skipped,
           jamb:jamb, maxScore:questions.length*SC_CORRECT, rawPct:rawPct };
}

function computeTopicAnalysis(){
  var bySubject={}, byTopic={}, byTaxonomy={};
  function tally(map,key,correct,wrong){
    if (!key) return;
    if (!map[key]) map[key]={correct:0,wrong:0,skipped:0,total:0};
    map[key].total++;
    if (correct) map[key].correct++;
    else if (wrong) map[key].wrong++;
    else map[key].skipped++;
  }
  questions.forEach(function(q,i){
    var a=userAnswers[i], correct=a&&a===q.answer, wrong=a&&a!==q.answer;
    tally(bySubject, q.subject||'General', correct, wrong);
    tally(byTopic, q.topic, correct, wrong);
    tally(byTaxonomy, q.taxonomy, correct, wrong);
  });
  function rank(map){
    return Object.keys(map).map(function(k){
      var d=map[k];
      return {name:k,correct:d.correct,wrong:d.wrong,skipped:d.skipped,total:d.total,
              pct:d.total?Math.round(d.correct/d.total*100):0};
    }).sort(function(a,b){return b.pct-a.pct;});
  }
  return {bySubject:rank(bySubject), byTopic:rank(byTopic), byTaxonomy:rank(byTaxonomy)};
}
